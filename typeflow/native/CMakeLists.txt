cmake_minimum_required(VERSION 3.20)
project(typeflow_bridge LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(typeflow_rime SHARED rime_bridge.cpp)
target_compile_features(typeflow_rime PRIVATE cxx_std_17)
target_include_directories(typeflow_rime PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../src")
target_link_libraries(typeflow_rime PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(typeflow_rime PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  target_compile_definitions(typeflow_rime PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  add_executable(typeflow_endpoint_smoke ../integration/weasel/endpoint_smoke.cpp)
  target_compile_features(typeflow_endpoint_smoke PRIVATE cxx_std_17)
  target_link_libraries(typeflow_endpoint_smoke PRIVATE comctl32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(typeflow_bridge_load_smoke bridge_load_smoke.cpp)
  target_compile_features(typeflow_bridge_load_smoke PRIVATE cxx_std_17)
  target_link_libraries(typeflow_bridge_load_smoke PRIVATE typeflow_rime)
  add_test(NAME bridge_load_failures COMMAND typeflow_bridge_load_smoke $<TARGET_FILE:typeflow_rime>)
endif()
if(MSVC)
  target_compile_options(typeflow_endpoint_smoke PRIVATE /W4 /WX /utf-8)
  target_compile_options(typeflow_rime PRIVATE /W4 /WX /utf-8)
  set_property(TARGET typeflow_rime PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(typeflow_rime PRIVATE -Wall -Wextra -Werror)
endif()
